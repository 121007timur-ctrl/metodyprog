45QString ClientAPI::sendAndReceive(const QString& message, int timeoutMs)
46{
47 if (!m_connected) return "ERROR: Not connected";
48
49 QByteArray data = message.toUtf8() + "\r\n";
50 m_socket->write(data);
51 if (!m_socket->waitForBytesWritten(timeoutMs)) return "ERROR: Send failed";
52 if (!m_socket->waitForReadyRead(timeoutMs)) return "ERROR: No response";
53
54 return QString::fromUtf8(m_socket->readAll());
